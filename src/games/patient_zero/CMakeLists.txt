# Optional CMake build (the Makefile is the primary build).
#   cmake -B build-cmake -DCMAKE_BUILD_TYPE=Release && cmake --build build-cmake -j
cmake_minimum_required(VERSION 3.16)
project(patient_zero C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

file(GLOB_RECURSE PZ_SOURCES CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/src/*.c)
add_executable(patient_zero ${PZ_SOURCES})
target_include_directories(patient_zero PRIVATE ${CMAKE_SOURCE_DIR}/src)
target_compile_definitions(patient_zero PRIVATE _DEFAULT_SOURCE)
target_compile_options(patient_zero PRIVATE -Wall -Wextra -Wshadow -Wno-unused-parameter -Wno-missing-field-initializers)

option(PZ_SANITIZE "Build with AddressSanitizer + UBSan" OFF)
if(PZ_SANITIZE)
  target_compile_options(patient_zero PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(patient_zero PRIVATE -fsanitize=address,undefined)
endif()

# raylib: prefer an installed CMake package, then pkg-config, then Homebrew prefixes
find_package(raylib 5.5 QUIET)
if(raylib_FOUND)
  target_link_libraries(patient_zero PRIVATE raylib)
else()
  find_package(PkgConfig QUIET)
  if(PkgConfig_FOUND)
    pkg_check_modules(RAYLIB QUIET raylib)
  endif()
  if(RAYLIB_FOUND)
    target_include_directories(patient_zero PRIVATE ${RAYLIB_INCLUDE_DIRS})
    target_link_directories(patient_zero PRIVATE ${RAYLIB_LIBRARY_DIRS})
    target_link_libraries(patient_zero PRIVATE ${RAYLIB_LIBRARIES})
  else()
    find_path(RAYLIB_INC raylib.h HINTS /opt/homebrew/include /usr/local/include)
    find_library(RAYLIB_LIB raylib HINTS /opt/homebrew/lib /usr/local/lib)
    if(NOT RAYLIB_INC OR NOT RAYLIB_LIB)
      message(FATAL_ERROR "raylib not found: brew install raylib")
    endif()
    target_include_directories(patient_zero PRIVATE ${RAYLIB_INC})
    target_link_libraries(patient_zero PRIVATE ${RAYLIB_LIB})
  endif()
endif()

if(APPLE)
  target_link_libraries(patient_zero PRIVATE "-framework CoreVideo" "-framework IOKit" "-framework Cocoa" "-framework GLUT" "-framework OpenGL")
elseif(UNIX)
  target_link_libraries(patient_zero PRIVATE GL m pthread dl rt X11)
endif()
